<iframe id="mapIframe" width="100%" height="500px" style="border:none;"></iframe>

<script>
    // 直接获取 map.txt 文件
    fetch('map.txt')
        .then(response => response.text())  // 读取文件内容为文本
        .then(content => {
            // 这里的 content 已经是完整的 HTML，所以我们可以直接将它设置为 iframe 的 srcdoc
            var iframe = document.getElementById('mapIframe');
            iframe.srcdoc = content;  // 使用 srcdoc 属性嵌入 HTML 内容
        })
        .catch(error => {
            console.error('Error fetching the map.txt file:', error);
        });
</script>
